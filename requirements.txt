# Runtime deps for app.py ONLY, deliberately kept loose.
#
# Hard-pinning every package makes pip's resolver backtrack, and numpy==1.26.4 has no
# wheels for Python 3.13 (it would compile from source, which is what stalls a build).
# Ranges let the resolver pick wheels for whatever Python the host runs.
#
# scikit-learn stays pinned: artifacts/ensemble_deploy.joblib is a 1.6.1 pickle and the
# version that loads it must match the version that wrote it.
streamlit>=1.34
altair>=5.0,<6
pandas>=2.0,<3
numpy>=1.26,<3
scikit-learn==1.6.1
xgboost>=2.0,<4
lightgbm>=4.0,<5
joblib>=1.3
